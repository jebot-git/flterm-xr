cmake_minimum_required(VERSION 3.24)
project(flterm-xr VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
find_package(PkgConfig REQUIRED)
find_package(FLTK CONFIG REQUIRED HINTS /opt/fltk-vr/toolkit/share/fltk)
find_package(nlohmann_json 3.2 REQUIRED)
pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET sdl2 freetype2 fontconfig zlib openxr egl gl xkbcommon)
add_library(flterm-core src/Fl_Term.cxx src/host.cxx xr/snapshot.cpp xr/pty.cpp xr/renderer.cpp xr/dome-font.cpp)
target_include_directories(flterm-core PUBLIC src xr)
target_compile_options(flterm-core PRIVATE -Wno-write-strings)
target_link_libraries(flterm-core PUBLIC fltk::fltk-shared PkgConfig::DEPS nlohmann_json::nlohmann_json util)
target_compile_definitions(flterm-core PRIVATE FLTERM_FONT_DIR="${CMAKE_INSTALL_FULL_DATADIR}/flterm-xr/fonts" FLTERM_BUILD_FONT="${CMAKE_CURRENT_SOURCE_DIR}/xr/fonts/unifont-17.0.04.bdf")
add_executable(flterm-xr xr/main.cpp xr/graphics.cpp xr/xr.cpp xr/keyboard.cpp xr/pointer.cpp xr/layout-state.cpp xr/clipboard.cpp)
target_link_libraries(flterm-xr PRIVATE flterm-core fltk::gl-shared)
include(xr/desktop-menu.cmake)
de_xr_desktop_menu(flterm-xr)
set_target_properties(flterm-xr PROPERTIES BUILD_RPATH "$<TARGET_FILE_DIR:fltk::fltk-shared>" INSTALL_RPATH "$<TARGET_FILE_DIR:fltk::fltk-shared>")
configure_file(xr/packaging/flterm-xr-launch.in flterm-xr-launch @ONLY)
file(CHMOD "${CMAKE_CURRENT_BINARY_DIR}/flterm-xr-launch" PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
configure_file(xr/packaging/org.domed.FLTerm.desktop.in org.domed.FLTerm.desktop @ONLY)
configure_file(xr/packaging/org.domed.FLTerm.XR.desktop.in org.domed.FLTerm.XR.desktop @ONLY)
install(TARGETS flterm-xr RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS xr/flterm-xrctl "${CMAKE_CURRENT_BINARY_DIR}/flterm-xr-launch" DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/org.domed.FLTerm.desktop" "${CMAKE_CURRENT_BINARY_DIR}/org.domed.FLTerm.XR.desktop" DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(DIRECTORY xr/fonts/ DESTINATION ${CMAKE_INSTALL_DATADIR}/flterm-xr/fonts)
install(FILES README.XR.md LICENSE xr/LICENSE.ghostty-xr xr/upstream.json DESTINATION ${CMAKE_INSTALL_DOCDIR})
include(CTest)
if(BUILD_TESTING)
 foreach(test layout layout-state keyboard pointer)
  set(extra)
  if(NOT test STREQUAL "layout")
   set(extra xr/${test}.cpp)
  endif()
  add_executable(${test}-test xr/tests/${test}.cpp ${extra})
  target_include_directories(${test}-test PRIVATE xr)
  target_compile_options(${test}-test PRIVATE -UNDEBUG)
  target_link_libraries(${test}-test PRIVATE PkgConfig::DEPS nlohmann_json::nlohmann_json)
  add_test(NAME ${test} COMMAND ${test}-test)
 endforeach()
 add_executable(control-strings-test xr/tests/control-strings.cpp)
 target_compile_options(control-strings-test PRIVATE -UNDEBUG)
 target_link_libraries(control-strings-test PRIVATE flterm-core)
 add_test(NAME control-strings COMMAND control-strings-test)
 add_executable(powerline-test xr/tests/powerline.cpp)
 target_compile_options(powerline-test PRIVATE -UNDEBUG)
 target_link_libraries(powerline-test PRIVATE flterm-core)
 add_test(NAME powerline COMMAND powerline-test)
 add_executable(core-test xr/tests/core.cpp)
 target_compile_options(core-test PRIVATE -UNDEBUG)
 target_link_libraries(core-test PRIVATE flterm-core)
 add_test(NAME core COMMAND core-test)
endif()
